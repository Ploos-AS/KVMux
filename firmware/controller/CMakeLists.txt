cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(kvmux_controller C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(kvmux_controller main.c)
target_link_libraries(kvmux_controller pico_stdlib)
pico_enable_stdio_usb(kvmux_controller 1)
pico_enable_stdio_uart(kvmux_controller 0)
pico_add_extra_outputs(kvmux_controller)
